     1→import type { Block, CustomAsset, EngineEvent, Project, Track } from "./types";
     2→import { getBlockType } from "./blocks";
     3→import { audioEngine } from "./audio";
     4→
     5→// ===== Futtató motor =====
     6→// Amikor a léptetőfej elér egy blokkot, a motor "kilövi" annak hatását:
     7→// entitást hoz létre, hangot szólaltat meg, változót ír, vagy ugrik.
     8→
     9→const W = 960;
    10→const H = 540;
    11→const VARS_KEY = "appmixer.vars.v1";
    12→
    13→type Role = "player" | "enemy" | "pickup" | "deco" | "text" | "projectile";
    14→
    15→interface Entity {
    16→  id: number;
    17→  kind: "ship" | "meteor" | "star" | "cloud" | "text" | "custom" | "projectile";
    18→  role: Role;
    19→  x: number; // 0..100 %
    20→  y: number; // 0..100 %
    21→  vx: number; // px/s
    22→  vy: number; // px/s
    23→  size: number; // px
    24→  color: string;
    25→  text?: string;
    26→  fontSize?: number;
    27→  image?: HTMLImageElement;
    28→  sticky: boolean;
    29→  life: number | null; // ms; null = örök
    30→  spin: boolean;
    31→  rotation: number;
    32→  spinSpeed: number;
    33→  bornAt: number;
    34→  poly: number[];
    35→}
    36→
    37→interface Particle {
    38→  x: number; y: number; vx: number; vy: number;
    39→  life: number; maxLife: number; color: string; size: number;
    40→}
    41→
    42→interface BgState {
    43→  type: "none" | "preset" | "color" | "image";
    44→  preset?: string;
    45→  color?: string;
    46→  image?: HTMLImageElement;
    47→}
    48→
    49→export class RuntimeEngine {
    50→  private canvas: HTMLCanvasElement | null = null;
    51→  private ctx: CanvasRenderingContext2D | null = null;
    52→  private raf = 0;
    53→  private lastFrame = 0;
    54→  private entityId = 1;
    55→
    56→  entities: Entity[] = [];
    57→  particles: Particle[] = [];
    58→  vars: Record<string, number> = { score: 0, lives: 3 };
    59→  private bg: BgState = { type: "none" };
    60→  private flash: { color: string; t: number } | null = null;
    61→  private shake = 0;
    62→  private invulnUntil = 0;
    63→  private lastShot = 0;
    64→  private gameOver = false;
    65→
    66→  playing = false;
    67→  paused = false;
    68→  step = 0;
    69→  private stepDeadline = 0;
    70→  private jumpGuard = 0;
    71→  private lastReportedStep = -1;
    72→
    73→  private keys = new Set<string>();
    74→  private keyboardUntil = 0;
    75→  private keyboardSpeed = 220;
    76→  private clickUntil = 0;
    77→  private clickMode: string = "shoot";
    78→
    79→  private getProject: () => Project;
    80→  private onEvent: (e: EngineEvent) => void;
    81→  private onStep: (s: number) => void;
    82→  private onGameOver: (score: number) => void;
    83→  private imageCache = new Map<string, HTMLImageElement>();
    84→  private starfield: { x: number; y: number; r: number; tw: number }[] = [];
    85→
    86→  constructor(
    87→    getProject: () => Project,
    88→    callbacks: {
    89→      onEvent: (e: EngineEvent) => void;
    90→      onStep: (s: number) => void;
    91→      onGameOver: (score: number) => void;
    92→    },
    93→  ) {
    94→    this.getProject = getProject;
    95→    this.onEvent = callbacks.onEvent;
    96→    this.onStep = callbacks.onStep;
    97→    this.onGameOver = callbacks.onGameOver;
    98→    if (typeof window !== "undefined") {
    99→      window.addEventListener("keydown", this.handleKeyDown);
   100→      window.addEventListener("keyup", this.handleKeyUp);
   101→      this.starfield = Array.from({ length: 90 }, () => ({
   102→        x: Math.random() * W,
   103→        y: Math.random() * H,
   104→        r: Math.random() * 1.6 + 0.4,
   105→        tw: Math.random() * Math.PI * 2,
   106→      }));
   107→      this.lastFrame = performance.now();
   108→      this.raf = requestAnimationFrame(this.loop);
   109→    }
   110→  }
   111→
   112→  attachCanvas(c: HTMLCanvasElement | null) {
   113→    this.canvas = c;
   114→    this.ctx = c ? c.getContext("2d") : null;
   115→  }
   116→
   117→  destroy() {
   118→    cancelAnimationFrame(this.raf);
   119→    window.removeEventListener("keydown", this.handleKeyDown);
   120→    window.removeEventListener("keyup", this.handleKeyUp);
   121→  }
   122→
   123→  // ---------- Irányítás ----------
   124→  play() {
   125→    const p = this.getProject();
   126→    audioEngine.ensureContext();
   127→    audioEngine.setMasterVolume(p.masterVolume);
   128→    this.entities = [];
   129→    this.particles = [];
   130→    this.vars = { score: 0, lives: 3 };
   131→    this.bg = { type: "none" };
   132→    this.flash = null;
   133→    this.shake = 0;
   134→    this.gameOver = false;
   135→    this.invulnUntil = 0;
   136→    this.keyboardUntil = 0;
   137→    this.clickUntil = 0;
   138→    this.step = 0;
   139→    this.jumpGuard = 0;
   140→    this.lastReportedStep = 0;
   141→    this.playing = true;
   142→    this.paused = false;
   143→    this.stepDeadline = performance.now();
   144→    this.emit(0, "Lejátszás indítása", "info");
   145→  }
   146→
   147→  pause() {
   148→    if (!this.playing) return;
   149→    this.paused = !this.paused;
   150→    if (!this.paused) this.stepDeadline = performance.now();
   151→    this.emit(this.step, this.paused ? "Szünet" : "Folytatás", "info");
   152→  }
   153→
   154→  stop() {
   155→    this.playing = false;
   156→    this.paused = false;
   157→    this.step = 0;
   158→    this.onStep(0);
   159→    audioEngine.stopAll();
   160→    this.emit(0, "Megállítva – a képernyő tartalma megmarad", "info");
   161→  }
   162→
   163→  resetView() {
   164→    this.entities = [];
   165→    this.particles = [];
   166→    this.bg = { type: "none" };
   167→    this.flash = null;
   168→    this.gameOver = false;
   169→    this.vars = { score: 0, lives: 3 };
   170→  }
   171→
   172→  private handleKeyDown = (e: KeyboardEvent) => {
   173→    if (!this.playing || this.paused) return;
   174→    const tag = (e.target as HTMLElement | null)?.tagName;
   175→    if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return;
   176→    if (["ArrowUp", "ArrowDown", "ArrowLeft", "ArrowRight", " ", "w", "a", "s", "d"].includes(e.key)) {
   177→      this.keys.add(e.key.length === 1 ? e.key.toLowerCase() : e.key);
   178→      if (e.key === " ") e.preventDefault();
   179→    }
   180→  };
   181→
   182→  private handleKeyUp = (e: KeyboardEvent) => {
   183→    const k = e.key.length === 1 ? e.key.toLowerCase() : e.key;
   184→    this.keys.delete(k);
   185→  };
   186→
   187→  handleCanvasClick(xPct: number, yPct: number) {
   188→    if (!this.playing || this.paused || performance.now() > this.clickUntil) return;
   189→    if (this.clickMode === "shoot") {
   190→      this.shootAt(xPct, yPct);
   191→    } else if (this.clickMode === "star") {
   192→      this.spawn({
   193→        kind: "star", role: "pickup", x: xPct, y: yPct, size: 18, color: "#fde047",
   194→        sticky: false, life: 6000, value: 5,
   195→      });
   196→      this.emit(this.step, "Csillag a kattintásnál", "gfx");
   197→    } else if (this.clickMode === "meteor") {
   198→      this.spawn({
   199→        kind: "meteor", role: "enemy", x: xPct, y: Math.max(2, yPct - 8), vy: 70, size: 24, color: "#9ca3af",
   200→        sticky: false, life: null, spin: true,
   201→      });
   202→      this.emit(this.step, "Meteor a kattintásnál", "gfx");
   203→    }
   204→  }
   205→
   206→  // ---------- Entitás segédek ----------
   207→  private spawn(p: Partial<Entity> & { kind: Entity["kind"]; role: Role; x: number; y: number }) {
   208→    if (this.entities.length > 300) this.entities.shift();
   209→    const e: Entity = {
   210→      id: this.entityId++,
   211→      kind: p.kind,
   212→      role: p.role,
   213→      x: p.x, y: p.y,
   214→      vx: p.vx ?? 0, vy: p.vy ?? 0,
   215→      size: p.size ?? 24,
   216→      color: p.color ?? "#ffffff",
   217→      text: p.text,
   218→      fontSize: p.fontSize,
   219→      image: p.image,
   220→      sticky: p.sticky ?? false,
   221→      life: p.life ?? null,
   222→      spin: p.spin ?? false,
   223→      rotation: Math.random() * Math.PI * 2,
   224→      spinSpeed: (Math.random() - 0.5) * 4,
   225→      bornAt: performance.now(),
   226→      poly: Array.from({ length: 8 }, () => 0.75 + Math.random() * 0.5),
   227→    };
   228→    this.entities.push(e);
   229→    return e;
   230→  }
   231→
   232→  private getPlayer(): Entity | undefined {
   233→    return this.entities.find((e) => e.role === "player");
   234→  }
   235→
   236→  private shootAt(xPct: number, yPct: number) {
   237→    const now = performance.now();
   238→    if (now - this.lastShot < 140) return;
   239→    this.lastShot = now;
   240→    const p = this.getPlayer();
   241→    const sx = p ? p.x : 50;
   242→    const sy = p ? p.y - 4 : 95;
   243→    const dx = (xPct / 100) * W - (sx / 100) * W;
   244→    const dy = (yPct / 100) * H - (sy / 100) * H;
   245→    const len = Math.hypot(dx, dy) || 1;
   246→    const speed = 520;
   247→    this.spawn({
   248→      kind: "projectile", role: "projectile", x: sx, y: sy,
   249→      vx: (dx / len) * speed, vy: (dy / len) * speed,
   250→      size: 10, color: "#fbbf24", sticky: false, life: 2500,
   251→    });
   252→    audioEngine.playSfx("sfx.laser", 0.7);
   253→  }
   254→
   255→  private burst(x: number, y: number, count: number, color: string, speed = 180) {
   256→    for (let i = 0; i < count; i++) {
   257→      const a = Math.random() * Math.PI * 2;
   258→      const s = speed * (0.3 + Math.random() * 0.9);
   259→      this.particles.push({
   260→        x, y,
   261→        vx: Math.cos(a) * s, vy: Math.sin(a) * s,
   262→        life: 500 + Math.random() * 500, maxLife: 1000,
   263→        color, size: 2 + Math.random() * 3.5,
   264→      });
   265→    }
   266→    if (this.particles.length > 600) this.particles.splice(0, this.particles.length - 600);
   267→  }
   268→
   269→  private emit(step: number, label: string, tone: EngineEvent["tone"]) {
   270→    this.onEvent({ step, label, tone });
   271→  }
   272→
   273→  // ---------- Léptető ----------
   274→  private loop = () => {
   275→    const now = performance.now();
   276→    const dt = Math.min(50, now - this.lastFrame) / 1000;
   277→    this.lastFrame = now;
   278→
   279→    if (this.playing && !this.paused && !this.gameOver) {
   280→      const p = this.getProject();
   281→      while (now >= this.stepDeadline) {
   282→        this.fireStep(this.step, p);
   283→        this.step++;
   284→        this.stepDeadline += p.tempoMs;
   285→        if (this.step >= p.stepCount) {
   286→          if (p.loop) {
   287→            this.step = 0;
   288→            this.emit(0, "Körülfordulás – újra az elejéről", "logic");
   289→          } else {
   290→            this.playing = false;
   291→            this.emit(this.step, "A kompozíció véget ért", "info");
   292→            break;
   293→          }
   294→        }
   295→      }
   296→      if (this.step !== this.lastReportedStep) {
   297→        this.lastReportedStep = this.step;
   298→        this.onStep(this.step);
   299→      }
   300→    }
   301→
   302→    this.update(dt, now);
   303→    this.render(now);
   304→    this.raf = requestAnimationFrame(this.loop);
   305→  };
   306→
   307→  private fireStep(step: number, p: Project) {
   308→    const anySolo = p.tracks.some((t) => t.solo);
   309→    for (const track of p.tracks) {
   310→      const muted = track.muted || (anySolo && !track.solo);
   311→      if (muted) continue;
   312→      for (const b of p.blocks) {
   313→        if (b.trackId !== track.id || b.start !== step) continue;
   314→        try {
   315→          this.applyBlock(b, track, p);
   316→        } catch {
   317→          this.emit(step, "Hiba egy blokk futtatásakor", "warn");
   318→        }
   319→      }
   320→    }
   321→  }
   322→
   323→  private findAsset(p: Project, id: string): CustomAsset | undefined {
   324→    return p.assets.find((a) => a.id === id);
   325→  }
   326→
   327→  private getImage(asset: CustomAsset): HTMLImageElement | undefined {
   328→    let img = this.imageCache.get(asset.id);
   329→    if (!img) {
   330→      img = new Image();
   331→      img.src = asset.dataUrl;
   332→      this.imageCache.set(asset.id, img);
   333→    }
   334→    return img.complete && img.naturalWidth > 0 ? img : undefined;
   335→  }
   336→
   337→  private applyBlock(b: Block, track: Track, p: Project) {
   338→    const isCustom = b.typeId.startsWith("custom:");
   339→    const def = getBlockType(b.typeId);
   340→    const assetId = isCustom ? b.typeId.slice(7) : undefined;
   341→    const asset = assetId ? this.findAsset(p, assetId) : undefined;
   342→    const kind = isCustom ? (asset?.kind === "audio" ? "customAudio" : "customImage") : def?.kind;
   343→    const pr = b.params;
   344→    const vol = track.volume;
   345→
   346→    switch (kind) {
   347→      case "bg": {
   348→        if (b.typeId === "gfx.bg.preset") {
   349→          this.bg = { type: "preset", preset: String(pr.preset ?? "space") };
   350→          this.emit(b.start, `Háttér: ${String(pr.preset)}`, "gfx");
   351→        } else {
   352→          this.bg = { type: "color", color: String(pr.color ?? "#1e1b4b") };
   353→          this.emit(b.start, "Háttérszín beállítva", "gfx");
   354→        }
   355→        break;
   356→      }
   357→      case "sprite": {
   358→        if (b.typeId === "gfx.sprite.ship") {
   359→          this.entities = this.entities.filter((e) => e.role !== "player");
   360→          this.spawn({
   361→            kind: "ship", role: "player",
   362→            x: Number(pr.x ?? 50), y: Number(pr.y ?? 82),
   363→            size: Number(pr.size ?? 34), color: String(pr.color ?? "#fbbf24"),
   364→            sticky: true, life: null,
   365→          });
   366→          this.emit(b.start, "Űrhajó (játékos) kész", "gfx");
   367→        } else if (b.typeId === "gfx.sprite.meteor") {
   368→          this.spawn({
   369→            kind: "meteor", role: "enemy",
   370→            x: Number(pr.x ?? 50), y: -4,
   371→            vy: Number(pr.vy ?? 55), size: Number(pr.size ?? 26),
   372→            color: "#9ca3af", spin: Boolean(pr.spin), sticky: false, life: null,
   373→          });
   374→          this.emit(b.start, `Meteor hullik (x${pr.x})`, "gfx");
   375→        } else if (b.typeId === "gfx.sprite.star") {
   376→          this.spawn({
   377→            kind: "star", role: "pickup",
   378→            x: Number(pr.x ?? 50), y: Number(pr.y ?? 40),
   379→            size: Number(pr.size ?? 18), color: "#fde047",
   380→            sticky: true, life: null,
   381→          });
   382→          this.emit(b.start, "Csillag vár gyűjtésre", "gfx");
   383→        } else if (b.typeId === "gfx.sprite.cloud") {
   384→          this.spawn({
   385→            kind: "cloud", role: "deco",
   386→            x: Number(pr.x ?? 20), y: Number(pr.y ?? 25),
   387→            vx: Number(pr.vx ?? 12), size: Number(pr.size ?? 36),
   388→            color: String(pr.color ?? "#e2e8f0"), sticky: true, life: null,
   389→          });
   390→        }
   391→        break;
   392→      }
   393→      case "text": {
   394→        this.spawn({
   395→          kind: "text", role: "text",
   396→          x: Number(pr.x ?? 50), y: Number(pr.y ?? 20),
   397→          text: String(pr.text ?? "Szöveg"), fontSize: Number(pr.size ?? 28),
   398→          color: String(pr.color ?? "#ffffff"),
   399→          sticky: Boolean(pr.sticky), life: pr.sticky ? null : b.length * p.tempoMs + 400,
   400→          size: Number(pr.size ?? 28),
   401→        });
   402→        this.emit(b.start, `Szöveg: „${String(pr.text ?? "").slice(0, 24)}”`, "gfx");
   403→        break;
   404→      }
   405→      case "effect": {
   406→        if (b.typeId === "gfx.effect.explosion") {
   407→          this.burst((Number(pr.x) / 100) * W, (Number(pr.y) / 100) * H, Number(pr.count ?? 26), String(pr.color ?? "#fb923c"));
   408→        } else if (b.typeId === "gfx.effect.flash") {
   409→          this.flash = { color: String(pr.color ?? "#ffffff"), t: Number(pr.strength ?? 60) / 100 };
   410→        } else if (b.typeId === "gfx.effect.fireworks") {
   411→          const n = Number(pr.count ?? 3);
   412→          for (let i = 0; i < n; i++) {
   413→            const fx = W * (0.15 + Math.random() * 0.7);
   414→            const fy = H * (0.15 + Math.random() * 0.5);
   415→            const hue = Math.floor(Math.random() * 360);
   416→            setTimeout(() => this.burst(fx, fy, 34, `hsl(${hue} 90% 65%)`, 240), i * 350);
   417→          }
   418→        }
   419→        break;
   420→      }
   421→      case "sfx": {
   422→        audioEngine.playSfx(b.typeId, Number(pr.volume ?? 70) / 100, vol);
   423→        this.emit(b.start, `Hang: ${def?.label ?? b.typeId}`, "sfx");
   424→        break;
   425→      }
   426→      case "music": {
   427→        audioEngine.playMusic(b.typeId, pr, b.length, p.tempoMs, Number(pr.volume ?? 70) / 100, vol);
   428→        this.emit(b.start, `Zene: ${def?.label ?? b.typeId} (${b.length} lépés)`, "sfx");
   429→        break;
   430→      }
   431→      case "keyboard": {
   432→        this.keyboardUntil = performance.now() + b.length * p.tempoMs;
   433→        this.keyboardSpeed = Number(pr.speed ?? 220);
   434→        this.emit(b.start, "Billentyű-irányítás aktív (nyilak/WASD, Space=lövés)", "info");
   435→        break;
   436→      }
   437→      case "click": {
   438→        this.clickUntil = performance.now() + b.length * p.tempoMs;
   439→        this.clickMode = String(pr.mode ?? "shoot");
   440→        this.emit(b.start, `Kattintás-esemény aktív (${this.clickMode})`, "info");
   441→        break;
   442→      }
   443→      case "counter": {
   444→        const name = String(pr.variable ?? "kor");
   445→        this.vars[name] = (this.vars[name] ?? 0) + Number(pr.delta ?? 1);
   446→        this.emit(b.start, `${name} = ${this.vars[name]}`, "data");
   447→        break;
   448→      }
   449→      case "check": {
   450→        let cond = true;
   451→        let desc = "";
   452→        if (b.typeId === "logic.random") {
   453→          cond = Math.random() * 100 < Number(pr.chance ?? 50);
   454→          desc = `véletlen (${pr.chance}%) ${cond ? "bekövetkezett" : "nem történt"}`;
   455→        } else {
   456→          const val = this.vars[String(pr.variable ?? "score")] ?? 0;
   457→          const target = Number(pr.value ?? 0);
   458→          const op = String(pr.op ?? ">=");
   459→          cond = op === ">=" ? val >= target : op === "<=" ? val <= target : Math.abs(val - target) < 1e-9;
   460→          desc = `${pr.variable}=${val} ${op} ${target} → ${cond ? "igaz" : "hamis"}`;
   461→        }
   462→        if (cond) {
   463→          const action = String(pr.action ?? "text");
   464→          const av = String(pr.actionValue ?? "");
   465→          if (action === "text") {
   466→            this.spawn({
   467→              kind: "text", role: "text", x: Number(pr.x ?? 50), y: Number(pr.y ?? 50),
   468→              text: av, fontSize: 34, color: "#a3e635", sticky: false, life: 2600, size: 34,
   469→            });
   470→          } else if (action === "jump") {
   471→            this.jumpTo(parseInt(av, 10) || 0);
   472→          } else if (action === "sound") {
   473→            const map: Record<string, string> = { laser: "sfx.laser", robbanas: "sfx.explosion", erme: "sfx.coin", pitty: "sfx.blip", suhintas: "sfx.swoosh", riasztas: "sfx.alarm" };
   474→            audioEngine.playSfx(map[av] ?? "sfx.blip", 0.8, vol);
   475→          } else if (action === "clear") {
   476→            this.entities = this.entities.filter((e) => e.role === "player");
   477→          }
   478→        }
   479→        this.emit(b.start, `Feltétel: ${desc}`, "logic");
   480→        break;
   481→      }
   482→      case "jump": {
   483→        this.jumpTo(Number(pr.target ?? 0));
   484→        this.emit(b.start, `Ugrás a ${pr.target}. lépésre`, "logic");
   485→        break;
   486→      }
   487→      case "clear": {
   488→        const mode = String(pr.mode ?? "enemies");
   489→        if (mode === "all") this.entities = this.entities.filter((e) => e.role === "player");
   490→        else if (mode === "enemies") this.entities = this.entities.filter((e) => e.role !== "enemy");
   491→        else this.entities = this.entities.filter((e) => e.role === "player" || e.role === "enemy");
   492→        this.emit(b.start, `Takarítás (${mode})`, "logic");
   493→        break;
   494→      }
   495→      case "setvar": {
   496→        const name = b.typeId === "data.lives" ? "lives" : String(pr.variable ?? "score");
   497→        this.vars[name] = Number(pr.value ?? 0);
   498→        this.emit(b.start, `${name} = ${pr.value}`, "data");
   499→        break;
   500→      }
   501→      case "save": {
   502→        try { localStorage.setItem(VARS_KEY, JSON.stringify(this.vars)); } catch {}
   503→        this.emit(b.start, "Változók elmentve a böngészőbe", "data");
   504→        break;
   505→      }
   506→      case "load": {
   507→        try {
   508→          const raw = localStorage.getItem(VARS_KEY);
   509→          if (raw) this.vars = { ...this.vars, ...JSON.parse(raw) };
   510→          this.emit(b.start, "Változók betöltve", "data");
   511→        } catch {}
   512→        break;
   513→      }
   514→      case "customImage": {
   515→        if (!asset) { this.emit(b.start, "Hiányzó feltöltött kép", "warn"); break; }
   516→        const img = this.getImage(asset);
   517→        const mode = String(pr.mode ?? "deco");
   518→        if (mode === "bg") {
   519→          if (img) this.bg = { type: "image", image: img };
   520→          this.emit(b.start, `Háttérkép: ${asset.name}`, "gfx");
   521→        } else {
   522→          const role: Role = mode === "player" ? "player" : mode === "enemy" ? "enemy" : mode === "pickup" ? "pickup" : "deco";
   523→          if (role === "player") this.entities = this.entities.filter((e) => e.role !== "player");
   524→          this.spawn({
   525→            kind: "custom", role,
   526→            x: Number(pr.x ?? 50), y: Number(pr.y ?? 50),
   527→            vx: Number(pr.vx ?? 0), vy: Number(pr.vy ?? 0),
   528→            size: Number(pr.size ?? 40), image: img,
   529→            sticky: Boolean(pr.sticky ?? true), life: Boolean(pr.sticky ?? true) ? null : b.length * p.tempoMs + 400,
   530→          });
   531→          this.emit(b.start, `Kép megjelenítve: ${asset.name} (${mode})`, "gfx");
   532→        }
   533→        break;
   534→      }
   535→      case "customAudio": {
   536→        if (!asset) { this.emit(b.start, "Hiányzó feltöltött hang", "warn"); break; }
   537→        audioEngine.playDataUrl(asset.dataUrl, Number(pr.volume ?? 80) / 100);
   538→        this.emit(b.start, `Saját hang: ${asset.name}`, "sfx");
   539→        break;
   540→      }
   541→    }

   544→  private jumpTo(step: number) {
   545→    if (this.jumpGuard > 40) return; // örök ciklus védő
   546→    this.jumpGuard++;
   547→    setTimeout(() => (this.jumpGuard = 0), 1500);
   548→    this.step = Math.max(0, Math.min(this.getProject().stepCount - 1, step));
   549→    this.stepDeadline = performance.now() + this.getProject().tempoMs;
   550→    this.lastReportedStep = this.step;
   551→    this.onStep(this.step);
   552→  }

   554→  // ---------- Frissítés ----------
   555→  private update(dt: number, now: number) {
   556→    const player = this.getPlayer();
   557→    const kbActive = now < this.keyboardUntil;
   558→
   559→    // Játékos mozgása
   560→    if (player && kbActive && !this.gameOver) {
   561→      let dx = 0, dy = 0;
   562→      if (this.keys.has("ArrowLeft") || this.keys.has("a")) dx -= 1;
   563→      if (this.keys.has("ArrowRight") || this.keys.has("d")) dx += 1;
   564→      if (this.keys.has("ArrowUp") || this.keys.has("w")) dy -= 1;
   565→      if (this.keys.has("ArrowDown") || this.keys.has("s")) dy += 1;
   566→      player.x += (dx * this.keyboardSpeed * dt / W) * 100;
   567→      player.y += (dy * this.keyboardSpeed * dt / H) * 100;
   568→      player.x = Math.max(2, Math.min(98, player.x));
   569→      player.y = Math.max(4, Math.min(96, player.y));
   570→      if (this.keys.has(" ")) this.shootAt(player.x, player.y - 30);
   571→    }
   572→
   573→    // Entitások mozgása
   574→    for (const e of this.entities) {
   575→      if (e.role === "player") continue;
   576→      e.x += (e.vx * dt / W) * 100;
   577→      e.y += (e.vy * dt / H) * 100;
   578→      if (e.spin) e.rotation += e.spinSpeed * dt;
   579→      if (e.kind === "cloud" || e.role === "deco") {
   580→        if (e.x > 108) e.x = -8;
   581→        if (e.x < -8) e.x = 108;
   582→      }
   583→      if (e.life !== null) {
   584→        e.life -= dt * 1000;
   585→      }
   586→    }
   587→
   588→    // Lejárt entitások törlése
   589→    this.entities = this.entities.filter((e) => {
   590→      if (e.life !== null && e.life <= 0) return false;
   591→      if (e.role === "enemy" && (e.y * H / 100) > H + 80) return false;
   592→      if (e.role === "projectile" && (e.y < -10 || e.x < -10 || e.x > 110 || (e.y * H / 100) > H + 10)) return false;
   593→      return true;
   594→    });
   595→
   596→    // Ütközések
   597→    if (!this.gameOver) {
   598→      const projectiles = this.entities.filter((e) => e.role === "projectile");
   599→      for (const proj of [...projectiles]) {
   600→        for (const enemy of this.entities.filter((e) => e.role === "enemy")) {
   601→          if (this.hit(proj, enemy, 0.9)) {
   602→            this.entities = this.entities.filter((x) => x !== proj && x !== enemy);
   603→            this.vars["score"] = (this.vars["score"] ?? 0) + 10;
   604→            this.burst((enemy.x / 100) * W, (enemy.y / 100) * H, 22, "#fb923c", 200);
   605→            audioEngine.playSfx("sfx.explosion", 0.55);
   606→            this.emit(this.step, "Találat! +10 pont", "logic");
   607→            break;
   608→          }
   609→        }
   610→      }
   611→      if (player && now > this.invulnUntil) {
   612→        for (const enemy of this.entities.filter((e) => e.role === "enemy")) {
   613→          if (this.hit(player, enemy, 0.78)) {
   614→            this.entities = this.entities.filter((x) => x !== enemy);
   615→            this.vars["lives"] = Math.max(0, (this.vars["lives"] ?? 3) - 1);
   616→            this.invulnUntil = now + 1200;
   617→            this.flash = { color: "#ef4444", t: 0.5 };
   618→            this.shake = 12;
   619→            this.burst((enemy.x / 100) * W, (enemy.y / 100) * H, 30, "#ef4444", 240);
   620→            audioEngine.playSfx("sfx.explosion", 0.8);
   621→            this.emit(this.step, `Találat értek! Életek: ${this.vars["lives"]}`, "warn");
   622→            if (this.vars["lives"] <= 0) {
   623→              this.gameOver = true;
   624→              this.playing = false;
   625→              this.paused = false;
   626→              this.onGameOver(this.vars["score"] ?? 0);
   627→              this.emit(this.step, "JÁTÉK VÉGE", "warn");
   628→            }
   629→            break;
   630→          }
   631→        }
   632→        for (const pick of this.entities.filter((e) => e.role === "pickup")) {
   633→          if (this.hit(player, pick, 1.1)) {
   634→            this.entities = this.entities.filter((x) => x !== pick);
   635→            this.vars["score"] = (this.vars["score"] ?? 0) + 5;
   636→            this.burst((pick.x / 100) * W, (pick.y / 100) * H, 16, "#fde047", 120);
   637→            audioEngine.playSfx("sfx.coin", 0.7);
   638→            this.emit(this.step, "Csillag begyűjtve! +5 pont", "logic");
   639→          }
   640→        }
   641→      }
   642→    }
   643→
   644→    // Részecskék
   645→    for (const p of this.particles) {
   646→      p.x += p.vx * dt;
   647→      p.y += p.vy * dt;
   648→      p.vy += 60 * dt;
   649→      p.life -= dt * 1000;
   650→    }
   651→    this.particles = this.particles.filter((p) => p.life > 0);
   652→
   653→    if (this.shake > 0) this.shake = Math.max(0, this.shake - 40 * dt);
   654→    if (this.flash) this.flash.t = Math.max(0, this.flash.t - dt * 2.2);
   655→  }

   657→  private hit(a: Entity, b: Entity, factor: number): boolean {
   658→    const ax = (a.x / 100) * W, ay = (a.y / 100) * H;
   659→    const bx = (b.x / 100) * W, by = (b.y / 100) * H;
   660→    const r = ((a.size + b.size) / 2) * factor;
   661→    return Math.hypot(ax - bx, ay - by) < r;
   662→  }

   684→        case "ship":

   687→        case "meteor":
   688→          this.drawMeteor(ctx, e);
   689→          break;
   690→        case "star":
   691→          this.drawStar(ctx, e, now);

   723→          break;
   724→      }
   725→      ctx.restore();

   732→      ctx.fillStyle = p.color;

   734→      ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
   735→      ctx.fill();
   736→    }

  1072→  private shade(hex: string, percent: number): string {
  1073→    const m = hex.replace("#", "");
  1074→    if (m.length !== 6) return hex;
  1075→    const n = parseInt(m, 16);
  1076→    let r = (n >> 16) & 255, g = (n >> 8) & 255, b = n & 255;
  1077→    r = Math.min(255, Math.round(r + 255 * percent / 100));
  1078→    g = Math.min(255, Math.round(g + 255 * percent / 100));
  1079→    b = Math.min(255, Math.round(b + 255 * percent / 100));
  1080→    return `rgb(${r},${g},${b})`;
  1081→  }
