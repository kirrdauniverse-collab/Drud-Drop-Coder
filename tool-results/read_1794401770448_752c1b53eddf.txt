   172→   172→  private handleKeyDown = (e: KeyboardEvent) => {
   173→   173→    if (!this.playing || this.paused) return;
   174→   174→    const tag = (e.target as HTMLElement | null)?.tagName;
   175→   175→    if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return;
   176→   176→    if (["ArrowUp", "ArrowDown", "ArrowLeft", "ArrowRight", " ", "w", "a", "s", "d"].includes(e.key)) {
   177→   177→      this.keys.add(e.key.length === 1 ? e.key.toLowerCase() : e.key);
   178→   178→      if (e.key === " ") e.preventDefault();
   179→   179→    }
   180→   180→  };
   181→   181→
   182→   182→  private handleKeyUp = (e: KeyboardEvent) => {
   183→   183→    const k = e.key.length === 1 ? e.key.toLowerCase() : e.key;
   184→   184→    this.keys.delete(k);
   185→   185→  };

   187→   187→  handleCanvasClick(xPct: number, yPct: number) {

   190→   190→      this.shootAt(xPct, yPct);
   191→   191→    } else if (this.clickMode === "star") {
   192→   192→      this.spawn({
   193→   193→        kind: "star", role: "pickup", x: xPct, y: yPct, size: 18, color: "#fde047",
   194→   194→        sticky: false, life: 6000, value: 5,
   195→   195→      });
   196→   196→      this.emit(this.step, "Csillag a kattintásnál", "gfx");

   198→   198→      this.spawn({
   199→   199→        kind: "meteor", role: "enemy", x: xPct, y: Math.max(2, yPct - 8), vy: 70, size: 24, color: "#9ca3af",
   200→   200→        sticky: false, life: null, spin: true,
   201→   201→      });
   202→   202→      this.emit(this.step, "Meteor a kattintásnál", "gfx");
   203→   203→    }

   219→   219→      image: p.image,
   220→   220→      sticky: p.sticky ?? false,
   221→   221→      life: p.life ?? null,
   222→   222→      spin: p.spin ?? false,
   223→   223→      rotation: Math.random() * Math.PI * 2,
   224→   224→      spinSpeed: (Math.random() - 0.5) * 4,
   225→   225→      bornAt: performance.now(),
   226→   226→      poly: Array.from({ length: 8 }, () => 0.75 + Math.random() * 0.5),

   236→   236→  private shootAt(xPct: number, yPct: number) {
   237→   237→    const now = performance.now();
   238→   238→    if (now - this.lastShot < 140) return;
   239→   239→    this.lastShot = now;
   240→   240→    const p = this.getPlayer();
   241→   241→    const sx = p ? p.x : 50;
   242→   242→    const sy = p ? p.y - 4 : 95;
   243→   243→    const dx = (xPct / 100) * W - (sx / 100) * W;
   244→   244→    const dy = (yPct / 100) * H - (sy / 100) * H;
   245→   245→    const len = Math.hypot(dx, dy) || 1;
   246→   246→    const speed = 520;
   247→   247→    this.spawn({
   248→   248→      kind: "projectile", role: "projectile", x: sx, y: sy,
   249→   249→      vx: (dx / len) * speed, vy: (dy / len) * speed,
   250→   250→      size: 10, color: "#fbbf24", sticky: false, life: 2500,
   251→   251→    });
   252→   252→    audioEngine.playSfx("sfx.laser", 0.7);
   253→   253→  }

   255→   255→  private burst(x: number, y: number, count: number, color: string, speed = 180) {
   256→   256→    for (let i = 0; i < count; i++) {
   257→   257→      const a = Math.random() * Math.PI * 2;
   258→   258→      const s = speed * (0.3 + Math.random() * 0.9);
   259→   259→      this.particles.push({
   260→   260→        x, y,
   261→   261→        vx: Math.cos(a) * s, vy: Math.sin(a) * s,
   262→   262→        life: 500 + Math.random() * 500, maxLife: 1000,
   263→   263→        color, size: 2 + Math.random() * 3.5,
   264→   264→      });
   265→   265→    }
   266→   266→    if (this.particles.length > 600) this.particles.splice(0, this.particles.length - 600);
   267→   267→  }

   269→   269→  private emit(step: number, label: string, tone: EngineEvent["tone"]) {

   273→   273→  // ---------- Léptető ----------
   274→   274→  private loop = () => {
   275→   275→    const now = performance.now();
   276→   276→    const dt = Math.min(50, now - this.lastFrame) / 1000;
   277→   277→    this.lastFrame = now;
   278→   278→
   279→   279→    if (this.playing && !this.paused && !this.gameOver) {
   280→   280→      const p = this.getProject();
   281→   281→      while (now >= this.stepDeadline) {
   282→   282→        this.fireStep(this.step, p);
   283→   283→        this.step++;
   284→   284→        this.stepDeadline += p.tempoMs;
   285→   285→        if (this.step >= p.stepCount) {
   286→   286→          if (p.loop) {
   287→   287→            this.step = 0;
   288→   288→            this.emit(0, "Körülfordulás – újra az elejéről", "logic");
   289→   289→          } else {
   290→   290→            this.playing = false;
   291→   291→            this.emit(this.step, "A kompozíció véget ért", "info");
   292→   292→            break;
   293→   293→          }
   294→   294→        }
   295→   295→      }
   296→   296→      if (this.step !== this.lastReportedStep) {
   297→   297→        this.lastReportedStep = this.step;
   298→   298→        this.onStep(this.step);
   299→   299→      }
   300→   300→    }
   301→   301→
   302→   302→    this.update(dt, now);
   303→   303→    this.render(now);
   304→   304→    this.raf = requestAnimationFrame(this.loop);
   305→   305→  };
   306→   306→
   307→   307→  private fireStep(step: number, p: Project) {
   308→   308→    const anySolo = p.tracks.some((t) => t.solo);
   309→   309→    for (const track of p.tracks) {
   310→   310→      const muted = track.muted || (anySolo && !track.solo);
   311→   311→      if (muted) continue;
   312→   312→      for (const b of p.blocks) {
   313→   313→        if (b.trackId !== track.id || b.start !== step) continue;
   314→   314→        try {
   315→   315→          this.applyBlock(b, track, p);
   316→   316→        } catch {
   317→   317→          this.emit(step, "Hiba egy blokk futtatásakor", "warn");
   318→   318→        }
   319→   319→      }
   320→   320→    }
   321→   321→  }

   323→   323→  private findAsset(p: Project, id: string): CustomAsset | undefined {
   324→   324→    return p.assets.find((a) => a.id === id);

   327→   327→  private getImage(asset: CustomAsset): HTMLImageElement | undefined {
   328→   328→    let img = this.imageCache.get(asset.id);
   329→   329→    if (!img) {
   330→   330→      img = new Image();
   331→   331→      img.src = asset.dataUrl;
   332→   332→      this.imageCache.set(asset.id, img);
   333→   333→    }
   334→   334→    return img.complete && img.naturalWidth > 0 ? img : undefined;
   335→   335→  }
   336→   336→
   337→   337→  private applyBlock(b: Block, track: Track, p: Project) {
   338→   338→    const isCustom = b.typeId.startsWith("custom:");
   339→   339→    const def = getBlockType(b.typeId);
   340→   340→    const assetId = isCustom ? b.typeId.slice(7) : undefined;
   341→   341→    const asset = assetId ? this.findAsset(p, assetId) : undefined;
   342→   342→    const kind = isCustom ? (asset?.kind === "audio" ? "customAudio" : "customImage") : def?.kind;
   343→   343→    const pr = b.params;
   344→   344→    const vol = track.volume;
   345→   345→
   346→   346→    switch (kind) {
   347→   347→      case "bg": {
   348→   348→        if (b.typeId === "gfx.bg.preset") {
   349→   349→          this.bg = { type: "preset", preset: String(pr.preset ?? "space") };
   350→   350→          this.emit(b.start, `Háttér: ${String(pr.preset)}`, "gfx");
   351→   351→        } else {
   352→   352→          this.bg = { type: "color", color: String(pr.color ?? "#1e1b4b") };
   353→   353→          this.emit(b.start, "Háttérszín beállítva", "gfx");
   354→   354→        }
   355→   355→        break;
   356→   356→      }
   357→   357→      case "sprite": {
   358→   358→        if (b.typeId === "gfx.sprite.ship") {
   359→   359→          this.entities = this.entities.filter((e) => e.role !== "player");
   360→   360→          this.spawn({
   361→   361→            kind: "ship", role: "player",
   362→   362→            x: Number(pr.x ?? 50), y: Number(pr.y ?? 82),
   363→   363→            size: Number(pr.size ?? 34), color: String(pr.color ?? "#fbbf24"),
   364→   364→            sticky: true, life: null,
   365→   365→          });
   366→   366→          this.emit(b.start, "Űrhajó (játékos) kész", "gfx");
   367→   367→        } else if (b.typeId === "gfx.sprite.meteor") {
   368→   368→          this.spawn({
   369→   369→            kind: "meteor", role: "enemy",
   370→   370→            x: Number(pr.x ?? 50), y: -4,
   371→   371→            vy: Number(pr.vy ?? 55), size: Number(pr.size ?? 26),
   372→   372→            color: "#9ca3af", spin: Boolean(pr.spin), sticky: false, life: null,
   373→   373→          });
   374→   374→          this.emit(b.start, `Meteor hullik (x${pr.x})`, "gfx");
   375→   375→        } else if (b.typeId === "gfx.sprite.star") {
   376→   376→          this.spawn({
   377→   377→            kind: "star", role: "pickup",
   378→   378→            x: Number(pr.x ?? 50), y: Number(pr.y ?? 40),
   379→   379→            size: Number(pr.size ?? 18), color: "#fde047",
   380→   380→            sticky: true, life: null,
   381→   381→          });
   382→   382→          this.emit(b.start, "Csillag vár gyűjtésre", "gfx");
   383→   383→        } else if (b.typeId === "gfx.sprite.cloud") {
   384→   384→          this.spawn({
   385→   385→            kind: "cloud", role: "deco",
   386→   386→            x: Number(pr.x ?? 20), y: Number(pr.y ?? 25),
   387→   387→            vx: Number(pr.vx ?? 12), size: Number(pr.size ?? 36),
   388→   388→            color: String(pr.color ?? "#e2e8f0"), sticky: true, life: null,
   389→   389→          });
   390→   390→        }
   391→   391→        break;
   392→   392→      }
   393→   393→      case "text": {
   394→   394→        this.spawn({
   395→   395→          kind: "text", role: "text",
   396→   396→          x: Number(pr.x ?? 50), y: Number(pr.y ?? 20),
   397→   397→          text: String(pr.text ?? "Szöveg"), fontSize: Number(pr.size ?? 28),
   398→   398→          color: String(pr.color ?? "#ffffff"),
   399→   399→          sticky: Boolean(pr.sticky), life: pr.sticky ? null : b.length * p.tempoMs + 400,
   400→   400→          size: Number(pr.size ?? 28),
   401→   401→        });
   402→   402→        this.emit(b.start, `Szöveg: „${String(pr.text ?? "").slice(0, 24)}”`, "gfx");
   403→   403→        break;
   404→   404→      }
   405→   405→      case "effect": {
   406→   406→        if (b.typeId === "gfx.effect.explosion") {
   407→   407→          this.burst((Number(pr.x) / 100) * W, (Number(pr.y) / 100) * H, Number(pr.count ?? 26), String(pr.color ?? "#fb923c"));
   408→   408→        } else if (b.typeId === "gfx.effect.flash") {
   409→   409→          this.flash = { color: String(pr.color ?? "#ffffff"), t: Number(pr.strength ?? 60) / 100 };
   410→   410→        } else if (b.typeId === "gfx.effect.fireworks") {
   411→   411→          const n = Number(pr.count ?? 3);
   412→   412→          for (let i = 0; i < n; i++) {
   413→   413→            const fx = W * (0.15 + Math.random() * 0.7);
   414→   414→            const fy = H * (0.15 + Math.random() * 0.5);
   415→   415→            const hue = Math.floor(Math.random() * 360);
   416→   416→            setTimeout(() => this.burst(fx, fy, 34, `hsl(${hue} 90% 65%)`, 240), i * 350);
   417→   417→          }
   418→   418→        }
   419→   419→        break;
   420→   420→      }
   421→   421→      case "sfx": {
   422→   422→        audioEngine.playSfx(b.typeId, Number(pr.volume ?? 70) / 100, vol);
   423→   423→        this.emit(b.start, `Hang: ${def?.label ?? b.typeId}`, "sfx");
   424→   424→        break;
   425→   425→      }
   426→   426→      case "music": {
   427→   427→        audioEngine.playMusic(b.typeId, pr, b.length, p.tempoMs, Number(pr.volume ?? 70) / 100, vol);
   428→   428→        this.emit(b.start, `Zene: ${def?.label ?? b.typeId} (${b.length} lépés)`, "sfx");
   429→   429→        break;
   430→   430→      }
   431→   431→      case "keyboard": {
   432→   432→        this.keyboardUntil = performance.now() + b.length * p.tempoMs;

   555→   555→  private update(dt: number, now: number) {
   556→   556→    const player = this.getPlayer();
   557→   557→    const kbActive = now < this.keyboardUntil;
   558→   558→
   559→   559→    // Játékos mozgása
   560→   560→    if (player && kbActive && !this.gameOver) {
   561→   561→      let dx = 0, dy = 0;
   562→   562→      if (this.keys.has("ArrowLeft") || this.keys.has("a")) dx -= 1;
   563→   563→      if (this.keys.has("ArrowRight") || this.keys.has("d")) dx += 1;
   564→   564→      if (this.keys.has("ArrowUp") || this.keys.has("w")) dy -= 1;
   565→   565→      if (this.keys.has("ArrowDown") || this.keys.has("s")) dy += 1;
   566→   566→      player.x += (dx * this.keyboardSpeed * dt / W) * 100;
   567→   567→      player.y += (dy * this.keyboardSpeed * dt / H) * 100;
   568→   568→      player.x = Math.max(2, Math.min(98, player.x));
   569→   569→      player.y = Math.max(4, Math.min(96, player.y));
   570→   570→      if (this.keys.has(" ")) this.shootAt(player.x, player.y - 30);
   571→   571→    }
   572→   572→
   573→   573→    // Entitások mozgása
   574→   574→    for (const e of this.entities) {
   575→   575→      if (e.role === "player") continue;
   576→   576→      e.x += (e.vx * dt / W) * 100;
   577→   577→      e.y += (e.vy * dt / H) * 100;
   578→   578→      if (e.spin) e.rotation += e.spinSpeed * dt;
   579→   579→      if (e.kind === "cloud" || e.role === "deco") {
   580→   580→        if (e.x > 108) e.x = -8;
   581→   581→        if (e.x < -8) e.x = 108;
   582→   582→      }
   583→   583→      if (e.life !== null) {
   584→   584→        e.life -= dt * 1000;
   585→   585→      }
   586→   586→    }
   587→   587→
   588→   588→    // Lejárt entitások törlése
   589→   589→    this.entities = this.entities.filter((e) => {
   590→   590→      if (e.life !== null && e.life <= 0) return false;
   591→   591→      if (e.role === "enemy" && (e.y * H / 100) > H + 80) return false;
   592→   592→      if (e.role === "projectile" && (e.y < -10 || e.x < -10 || e.x > 110 || (e.y * H / 100) > H + 10)) return false;
   593→   593→      return true;
   594→   594→    });
   595→   595→
   596→   596→    // Ütközések
   597→   597→    if (!this.gameOver) {
   598→   598→      const projectiles = this.entities.filter((e) => e.role === "projectile");
   599→   599→      for (const proj of [...projectiles]) {
   600→   600→        for (const enemy of this.entities.filter((e) => e.role === "enemy")) {
   601→   601→          if (this.hit(proj, enemy, 0.9)) {
   602→   602→            this.entities = this.entities.filter((x) => x !== proj && x !== enemy);
   603→   603→            this.vars["score"] = (this.vars["score"] ?? 0) + 10;
   604→   604→            this.burst((enemy.x / 100) * W, (enemy.y / 100) * H, 22, "#fb923c", 200);
   605→   605→            audioEngine.playSfx("sfx.explosion", 0.55);
   606→   606→            this.emit(this.step, "Találat! +10 pont", "logic");
   607→   607→            break;
   608→   608→          }
   609→   609→        }
   610→   610→      }
   611→   611→      if (player && now > this.invulnUntil) {
   612→   612→        for (const enemy of this.entities.filter((e) => e.role === "enemy")) {
   613→   613→          if (this.hit(player, enemy, 0.78)) {
   614→   614→            this.entities = this.entities.filter((x) => x !== enemy);
   615→   615→            this.vars["lives"] = Math.max(0, (this.vars["lives"] ?? 3) - 1);
   616→   616→            this.invulnUntil = now + 1200;
   617→   617→            this.flash = { color: "#ef4444", t: 0.5 };
   618→   618→            this.shake = 12;
   619→   619→            this.burst((enemy.x / 100) * W, (enemy.y / 100) * H, 30, "#ef4444", 240);
   620→   620→            audioEngine.playSfx("sfx.explosion", 0.8);
   621→   621→            this.emit(this.step, `Találat értek! Életek: ${this.vars["lives"]}`, "warn");
   622→   622→            if (this.vars["lives"] <= 0) {
   623→   623→              this.gameOver = true;
   624→   624→              this.playing = false;
   625→   625→              this.paused = false;
   626→   626→              this.onGameOver(this.vars["score"] ?? 0);
   627→   627→              this.emit(this.step, "JÁTÉK VÉGE", "warn");
   628→   628→            }
   629→   629→            break;
   630→   630→          }
   631→   631→        }
   632→   632→        for (const pick of this.entities.filter((e) => e.role === "pickup")) {
   633→   633→          if (this.hit(player, pick, 1.1)) {
   634→   634→            this.entities = this.entities.filter((x) => x !== pick);
   635→   635→            this.vars["score"] = (this.vars["score"] ?? 0) + 5;
   636→   636→            this.burst((pick.x / 100) * W, (pick.y / 100) * H, 16, "#fde047", 120);
   637→   637→            audioEngine.playSfx("sfx.coin", 0.7);
   638→   638→            this.emit(this.step, "Csillag begyűjtve! +5 pont", "logic");
   639→   639→          }
   640→   640→        }
   641→   641→      }
   642→   642→    }
   643→   643→
   644→   644→    // Részecskék
   645→   645→    for (const p of this.particles) {
   646→   646→      p.x += p.vx * dt;
   647→   647→      p.y += p.vy * dt;
   648→   648→      p.vy += 60 * dt;
   649→   649→      p.life -= dt * 1000;
   650→   650→    }
   651→   651→    this.particles = this.particles.filter((p) => p.life > 0);
   652→   652→
   653→   653→    if (this.shake > 0) this.shake = Math.max(0, this.shake - 40 * dt);
   654→   654→    if (this.flash) this.flash.t = Math.max(0, this.flash.t - dt * 2.2);
   655→   655→  }

   657→   657→  private hit(a: Entity, b: Entity, factor: number): boolean {
   658→   658→    const ax = (a.x / 100) * W, ay = (a.y / 100) * H;
   659→   659→    const bx = (b.x / 100) * W, by = (b.y / 100) * H;
   660→   660→    const r = ((a.size + b.size) / 2) * factor;
   661→   661→    return Math.hypot(ax - bx, ay - by) < r;
   662→   662→  }

   803→   803→    const s = e.size;
   804→   804→    ctx.rotate(e.rotation);
   805→   805→    ctx.fillStyle = "#78716c";
   806→   806→    ctx.strokeStyle = "#a8a29e";

  1072→  1072→  private shade(hex: string, percent: number): string {
  1073→  1073→    const m = hex.replace("#", "");
  1074→  1074→    if (m.length !== 6) return hex;
  1075→  1075→    const n = parseInt(m, 16);
  1076→  1076→    let r = (n >> 16) & 255, g = (n >> 8) & 255, b = n & 255;
  1077→  1077→    r = Math.min(255, Math.round(r + 255 * percent / 100));
  1078→  1078→    g = Math.min(255, Math.round(g + 255 * percent / 100));
  1079→  1079→    b = Math.min(255, Math.round(b + 255 * percent / 100));
  1080→  1080→    return `rgb(${r},${g},${b})`;
  1081→  1081→  }
